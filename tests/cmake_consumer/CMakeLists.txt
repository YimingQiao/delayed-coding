cmake_minimum_required(VERSION 3.20)
project(DelayedCodingConsumer LANGUAGES C CXX)
if(NOT DELAYED_CODING_SOURCE_DIR)
    message(FATAL_ERROR "Set DELAYED_CODING_SOURCE_DIR to the standalone checkout")
endif()
set(DELAYED_CODING_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DELAYED_CODING_SOURCE_DIR}" delayed-coding)
add_executable(consumer "${DELAYED_CODING_SOURCE_DIR}/examples/c_roundtrip.c")
target_link_libraries(consumer PRIVATE DelayedCoding::delayed_coding)
enable_testing()
add_test(NAME downstream_consumer COMMAND consumer)
